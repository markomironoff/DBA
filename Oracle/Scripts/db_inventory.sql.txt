/*
db_inventory.sql
run: sqlplus -s / as sysdba @db_inventory.sql

Non-CDB databases
CDB databases with PDBs
Single-instance databases
Oracle RAC databases
PDB count and status
SGA/PGA configuration per instance
Total CDB/non-CDB allocated storage
Temporary files
Redo logs
FRA
Character sets
Timezone file
TDE wallet status

*/


whenever sqlerror exit sql.sqlcode
whenever oserror exit failure

set sqlblanklines on
set echo off
set verify off
set feedback off
set heading on
set pagesize 50000
set linesize 32767
set trimspool on
set tab off
set termout off

define output_dir = '/tmp'

column database_name       new_value database_name       noprint
column inventory_timestamp new_value inventory_timestamp noprint

select lower(name) as database_name
from v$database;

select to_char(sysdate, 'YYYYMMDD_HH24MISS') as inventory_timestamp
from dual;

set markup csv on delimiter , quote on

spool &output_dir./&database_name._db_inventory_&inventory_timestamp..csv

with
params as
(
    select
        inst_id,
        max(case when name = 'sga_target'
                 then value end) as sga_target,
        max(case when name = 'sga_max_size'
                 then value end) as sga_max_size,
        max(case when name = 'pga_aggregate_target'
                 then value end) as pga_aggregate_target,
        max(case when name = 'pga_aggregate_limit'
                 then value end) as pga_aggregate_limit,
        max(case when name = 'memory_target'
                 then value end) as memory_target,
        max(case when name = 'memory_max_target'
                 then value end) as memory_max_target,
        max(case when name = 'cpu_count'
                 then value end) as cpu_count,
        max(case when name = 'processes'
                 then value end) as processes,
        max(case when name = 'sessions'
                 then value end) as sessions,
        max(case when name = 'db_block_size'
                 then value end) as db_block_size,
        max(case when name = 'compatible'
                 then value end) as compatible,
        max(case when name = 'cluster_database'
                 then value end) as cluster_database,
        max(case when name = 'filesystemio_options'
                 then value end) as filesystemio_options,
        max(case when name = 'disk_asynch_io'
                 then value end) as disk_asynch_io,
        max(case when name = 'db_writer_processes'
                 then value end) as db_writer_processes,
        max(case when name = 'db_recovery_file_dest'
                 then value end) as db_recovery_file_dest,
        max(case when name = 'db_recovery_file_dest_size'
                 then value end) as db_recovery_file_dest_size
    from gv$system_parameter
    where name in
    (
        'sga_target',
        'sga_max_size',
        'pga_aggregate_target',
        'pga_aggregate_limit',
        'memory_target',
        'memory_max_target',
        'cpu_count',
        'processes',
        'sessions',
        'db_block_size',
        'compatible',
        'cluster_database',
        'filesystemio_options',
        'disk_asynch_io',
        'db_writer_processes',
        'db_recovery_file_dest',
        'db_recovery_file_dest_size'
    )
    group by inst_id
),
sga_actual as
(
    select
        inst_id,
        sum(value) as sga_bytes
    from gv$sga
    group by inst_id
),
pga_actual as
(
    select
        inst_id,
        max(
            case
                when name = 'total PGA allocated'
                then value
            end
        ) as pga_allocated_bytes,
        max(
            case
                when name = 'maximum PGA allocated'
                then value
            end
        ) as pga_max_allocated_bytes
    from gv$pgastat
    where name in
    (
        'total PGA allocated',
        'maximum PGA allocated'
    )
    group by inst_id
),
instance_summary as
(
    select
        count(*) as instance_count
    from gv$instance
),
container_summary as
(
    select
        count(*) as container_count,
        sum(
            case
                when con_id > 2
                then 1
                else 0
            end
        ) as pdb_count,
        sum(
            case
                when con_id > 2
                 and open_mode = 'READ WRITE'
                then 1
                else 0
            end
        ) as pdb_read_write_count,
        sum(
            case
                when con_id > 2
                 and open_mode = 'READ ONLY'
                then 1
                else 0
            end
        ) as pdb_read_only_count,
        sum(
            case
                when con_id > 2
                 and open_mode = 'MOUNTED'
                then 1
                else 0
            end
        ) as pdb_mounted_count
    from v$containers
),
data_size as
(
    select
        count(*) as datafile_count,
        sum(bytes) as datafile_bytes,
        sum(
            case
                when autoextensible = 'YES'
                then greatest(bytes, maxbytes)
                else bytes
            end
        ) as datafile_maxbytes,
        sum(
            case
                when autoextensible = 'YES'
                then 1
                else 0
            end
        ) as autoextensible_datafile_count
    from cdb_data_files
),
temp_size as
(
    select
        count(*) as tempfile_count,
        sum(bytes) as tempfile_bytes,
        sum(
            case
                when autoextensible = 'YES'
                then greatest(bytes, maxbytes)
                else bytes
            end
        ) as tempfile_maxbytes,
        sum(
            case
                when autoextensible = 'YES'
                then 1
                else 0
            end
        ) as autoextensible_tempfile_count
    from cdb_temp_files
),
redo_size as
(
    select
        count(*) as redo_log_group_count,
        count(distinct thread#) as redo_thread_count,
        sum(bytes) as redo_bytes,
        min(bytes) as smallest_redo_log_bytes,
        max(bytes) as largest_redo_log_bytes
    from v$log
),
redo_members as
(
    select
        count(*) as redo_log_member_count
    from v$logfile
),
fra as
(
    select
        name,
        space_limit,
        space_used,
        space_reclaimable,
        number_of_files
    from v$recovery_file_dest
),
charset_info as
(
    select
        max(
            case
                when parameter = 'NLS_CHARACTERSET'
                then value
            end
        ) as database_charset,
        max(
            case
                when parameter = 'NLS_NCHAR_CHARACTERSET'
                then value
            end
        ) as national_charset
    from nls_database_parameters
),
timezone_info as
(
    select
        filename,
        version
    from v$timezone_file
),
controlfile_info as
(
    select
        count(*) as controlfile_count
    from v$controlfile
),
asm_storage as
(
    select
        case
            when sum(
                case
                    when name like '+%'
                    then 1
                    else 0
                end
            ) > 0
            then 'YES'
            else 'NO'
        end as uses_asm
    from v$datafile
),
wallet_info as
(
    select
        max(status) as wallet_status,
        max(wallet_type) as wallet_type
    from v$encryption_wallet
)
select
    sys_context('USERENV', 'SERVER_HOST')
                                                                as collection_host,
    d.name                                                      as database_name,
    d.db_unique_name                                            as db_unique_name,
    d.dbid                                                      as dbid,
    to_char(d.created, 'YYYY-MM-DD HH24:MI:SS')
                                                                as database_created,
    d.cdb                                                       as is_cdb,
    sys_context('USERENV', 'CON_NAME')                           as collection_container,

    ins.instance_count                                          as instance_count,
    i.instance_name                                             as instance_name,
    i.instance_number                                           as instance_number,
    i.thread#                                                   as thread_number,
    i.host_name                                                 as instance_host,
    i.version                                                   as oracle_version,
    i.version_full                                              as oracle_version_full,
    i.status                                                    as instance_status,
    i.database_status                                           as database_status,
    i.instance_role                                             as instance_role,
    i.parallel                                                  as rac_parallel,
    to_char(i.startup_time, 'YYYY-MM-DD HH24:MI:SS')
                                                                as startup_time,

    d.open_mode                                                 as open_mode,
    d.database_role                                             as database_role,
    d.log_mode                                                  as log_mode,
    d.force_logging                                             as force_logging,
    d.flashback_on                                              as flashback_on,
    d.protection_mode                                           as protection_mode,
    d.protection_level                                          as protection_level,
    d.switchover_status                                         as switchover_status,
    d.platform_name                                             as platform_name,

    p.compatible                                                as compatible,
    p.cluster_database                                          as cluster_database,
    p.cpu_count                                                 as cpu_count,
    p.processes                                                 as processes,
    p.sessions                                                  as sessions,
    p.db_block_size                                             as db_block_size_bytes,

    round(
        to_number(p.sga_target) / power(1024, 3),
        2
    )                                                           as sga_target_gb,
    round(
        to_number(p.sga_max_size) / power(1024, 3),
        2
    )                                                           as sga_max_size_gb,
    round(
        sa.sga_bytes / power(1024, 3),
        2
    )                                                           as current_sga_gb,

    round(
        to_number(p.pga_aggregate_target) / power(1024, 3),
        2
    )                                                           as pga_target_gb,
    round(
        to_number(p.pga_aggregate_limit) / power(1024, 3),
        2
    )                                                           as pga_limit_gb,
    round(
        pa.pga_allocated_bytes / power(1024, 3),
        2
    )                                                           as current_pga_gb,
    round(
        pa.pga_max_allocated_bytes / power(1024, 3),
        2
    )                                                           as maximum_pga_gb,

    round(
        to_number(p.memory_target) / power(1024, 3),
        2
    )                                                           as memory_target_gb,
    round(
        to_number(p.memory_max_target) / power(1024, 3),
        2
    )                                                           as memory_max_target_gb,

    cs.container_count                                          as container_count,
    cs.pdb_count                                                as pdb_count,
    cs.pdb_read_write_count                                     as pdb_read_write_count,
    cs.pdb_read_only_count                                      as pdb_read_only_count,
    cs.pdb_mounted_count                                        as pdb_mounted_count,

    ds.datafile_count                                           as datafile_count,
    ds.autoextensible_datafile_count                            as autoextend_datafile_count,
    round(
        ds.datafile_bytes / power(1024, 3),
        2
    )                                                           as datafile_allocated_gb,
    round(
        ds.datafile_bytes / power(1024, 4),
        2
    )                                                           as datafile_allocated_tb,
    round(
        ds.datafile_maxbytes / power(1024, 3),
        2
    )                                                           as datafile_max_gb,
    round(
        ds.datafile_maxbytes / power(1024, 4),
        2
    )                                                           as datafile_max_tb,

    ts.tempfile_count                                           as tempfile_count,
    ts.autoextensible_tempfile_count                            as autoextend_tempfile_count,
    round(
        ts.tempfile_bytes / power(1024, 3),
        2
    )                                                           as tempfile_allocated_gb,
    round(
        ts.tempfile_maxbytes / power(1024, 3),
        2
    )                                                           as tempfile_max_gb,

    rs.redo_thread_count                                        as redo_thread_count,
    rs.redo_log_group_count                                     as redo_log_group_count,
    rm.redo_log_member_count                                    as redo_log_member_count,
    round(
        rs.redo_bytes / power(1024, 3),
        2
    )                                                           as redo_log_total_gb,
    round(
        rs.smallest_redo_log_bytes / power(1024, 2),
        2
    )                                                           as smallest_redo_log_mb,
    round(
        rs.largest_redo_log_bytes / power(1024, 2),
        2
    )                                                           as largest_redo_log_mb,

    cf.controlfile_count                                        as controlfile_count,
    ast.uses_asm                                                as uses_asm,

    nvl(f.name, p.db_recovery_file_dest)                         as fra_destination,
    round(
        nvl(
            f.space_limit,
            to_number(p.db_recovery_file_dest_size)
        ) / power(1024, 3),
        2
    )                                                           as fra_limit_gb,
    round(
        f.space_used / power(1024, 3),
        2
    )                                                           as fra_used_gb,
    round(
        f.space_reclaimable / power(1024, 3),
        2
    )                                                           as fra_reclaimable_gb,
    f.number_of_files                                           as fra_file_count,

    ci.database_charset                                         as database_charset,
    ci.national_charset                                         as national_charset,
    ti.filename                                                 as timezone_file,
    ti.version                                                  as timezone_version,

    wi.wallet_status                                            as wallet_status,
    wi.wallet_type                                              as wallet_type,

    p.filesystemio_options                                      as filesystemio_options,
    p.disk_asynch_io                                            as disk_asynch_io,
    p.db_writer_processes                                       as db_writer_processes,

    to_char(
        systimestamp,
        'YYYY-MM-DD HH24:MI:SS TZH:TZM'
    )                                                           as collection_timestamp

from v$database d
cross join gv$instance i
join params p
    on p.inst_id = i.inst_id
left join sga_actual sa
    on sa.inst_id = i.inst_id
left join pga_actual pa
    on pa.inst_id = i.inst_id
cross join instance_summary ins
cross join container_summary cs
cross join data_size ds
cross join temp_size ts
cross join redo_size rs
cross join redo_members rm
cross join charset_info ci
cross join timezone_info ti
cross join controlfile_info cf
cross join asm_storage ast
cross join wallet_info wi
left join fra f
    on 1 = 1
order by
    i.instance_number;

spool off

set markup csv off
set termout on

prompt
prompt Inventory CSV created:
prompt &output_dir./&database_name._db_inventory_&inventory_timestamp..csv
prompt

exit success
