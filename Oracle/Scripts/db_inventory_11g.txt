whenever sqlerror exit sql.sqlcode
whenever oserror exit failure

set echo off
set feedback off
set verify off
set heading off
set pagesize 0
set linesize 32767
set trimspool on
set tab off
set sqlblanklines on
set termout off
set recsep off

column output_file new_value output_file noprint
select '/tmp/' || lower(name) || '_db_inventory_' || to_char(sysdate,'YYYYMMDD_HH24MISS') || '.csv' output_file from v$database;

spool &output_file
prompt "COLLECTION_HOST","DATABASE_NAME","DB_UNIQUE_NAME","DBID","DATABASE_CREATED","IS_CDB","INSTANCE_NAME","INSTANCE_NUMBER","THREAD_NUMBER","INSTANCE_HOST","ORACLE_VERSION","INSTANCE_STATUS","DATABASE_STATUS","INSTANCE_ROLE","RAC_PARALLEL","STARTUP_TIME","OPEN_MODE","DATABASE_ROLE","LOG_MODE","FORCE_LOGGING","FLASHBACK_ON","PROTECTION_MODE","PROTECTION_LEVEL","SWITCHOVER_STATUS","PLATFORM_NAME","COMPATIBLE","CLUSTER_DATABASE","CPU_COUNT","PROCESSES","SESSIONS","DB_BLOCK_SIZE_BYTES","SGA_TARGET_GB","SGA_MAX_SIZE_GB","CURRENT_SGA_GB","PGA_TARGET_GB","PGA_LIMIT_GB","CURRENT_PGA_GB","MAXIMUM_PGA_GB","MEMORY_TARGET_GB","MEMORY_MAX_TARGET_GB","CONTAINER_COUNT","PDB_COUNT","PDB_READ_WRITE_COUNT","PDB_READ_ONLY_COUNT","PDB_MOUNTED_COUNT","DATAFILE_COUNT","AUTOEXTEND_DATAFILE_COUNT","DATAFILE_ALLOCATED_GB","DATAFILE_ALLOCATED_TB","DATAFILE_MAX_GB","DATAFILE_MAX_TB","TEMPFILE_COUNT","AUTOEXTEND_TEMPFILE_COUNT","TEMPFILE_ALLOCATED_GB","TEMPFILE_MAX_GB","REDO_THREAD_COUNT","REDO_LOG_GROUP_COUNT","REDO_LOG_MEMBER_COUNT","REDO_LOG_TOTAL_GB","SMALLEST_REDO_LOG_MB","LARGEST_REDO_LOG_MB","CONTROLFILE_COUNT","USES_ASM","FRA_DESTINATION","FRA_LIMIT_GB","FRA_USED_GB","FRA_RECLAIMABLE_GB","FRA_FILE_COUNT","DATABASE_CHARSET","NATIONAL_CHARSET","TIMEZONE_FILE","TIMEZONE_VERSION","WALLET_STATUS","WALLET_TYPE","FILESYSTEMIO_OPTIONS","DISK_ASYNCH_IO","DB_WRITER_PROCESSES","COLLECTION_TIMESTAMP"

with p as (
  select inst_id,
    max(case when name='compatible' then value end) compatible,
    max(case when name='cluster_database' then value end) cluster_database,
    max(case when name='cpu_count' then value end) cpu_count,
    max(case when name='processes' then value end) processes,
    max(case when name='sessions' then value end) sessions,
    max(case when name='db_block_size' then value end) db_block_size,
    max(case when name='sga_target' then value end) sga_target,
    max(case when name='sga_max_size' then value end) sga_max_size,
    max(case when name='pga_aggregate_target' then value end) pga_target,
    max(case when name='memory_target' then value end) memory_target,
    max(case when name='memory_max_target' then value end) memory_max_target,
    max(case when name='filesystemio_options' then value end) filesystemio_options,
    max(case when name='disk_asynch_io' then value end) disk_asynch_io,
    max(case when name='db_writer_processes' then value end) db_writer_processes,
    max(case when name='db_recovery_file_dest' then value end) fra_dest,
    max(case when name='db_recovery_file_dest_size' then value end) fra_size
  from gv$system_parameter
  group by inst_id
), sa as (
  select inst_id,sum(value) bytes from gv$sga group by inst_id
), pa as (
  select inst_id,
    max(case when name='total PGA allocated' then value end) bytes,
    max(case when name='maximum PGA allocated' then value end) maxbytes
  from gv$pgastat group by inst_id
), df as (
  select count(*) cnt,sum(case when autoextensible='YES' then 1 else 0 end) autocnt,
    sum(bytes) bytes,sum(case when autoextensible='YES' then greatest(bytes,maxbytes) else bytes end) maxbytes
  from dba_data_files
), tf as (
  select count(*) cnt,sum(case when autoextensible='YES' then 1 else 0 end) autocnt,
    sum(bytes) bytes,sum(case when autoextensible='YES' then greatest(bytes,maxbytes) else bytes end) maxbytes
  from dba_temp_files
), rl as (
  select count(*) groups,count(distinct thread#) threads,sum(bytes) bytes,min(bytes) minbytes,max(bytes) maxbytes from v$log
), rm as (select count(*) members from v$logfile),
cf as (select count(*) cnt from v$controlfile),
ai as (select case when sum(case when name like '+%' then 1 else 0 end)>0 then 'YES' else 'NO' end uses_asm from v$datafile),
fr as (select name,space_limit,space_used,space_reclaimable,number_of_files from v$recovery_file_dest),
cs as (select max(case when parameter='NLS_CHARACTERSET' then value end) dbcs,max(case when parameter='NLS_NCHAR_CHARACTERSET' then value end) ncs from nls_database_parameters),
tz as (select filename,version from v$timezone_file)
select
'"'||replace(sys_context('USERENV','SERVER_HOST'),'"','""')||'","'||replace(d.name,'"','""')||'","'||replace(d.db_unique_name,'"','""')||'",'||d.dbid||',"'||to_char(d.created,'YYYY-MM-DD HH24:MI:SS')||'","NO","'||replace(i.instance_name,'"','""')||'",'||i.instance_number||','||i.thread#||',"'||replace(i.host_name,'"','""')||'","'||i.version||'","'||i.status||'","'||i.database_status||'","'||i.instance_role||'","'||i.parallel||'","'||to_char(i.startup_time,'YYYY-MM-DD HH24:MI:SS')||'","'||d.open_mode||'","'||d.database_role||'","'||d.log_mode||'","'||d.force_logging||'","'||d.flashback_on||'","'||d.protection_mode||'","'||d.protection_level||'","'||d.switchover_status||'","'||replace(d.platform_name,'"','""')||'","'||nvl(p.compatible,'')||'","'||nvl(p.cluster_database,'')||'",'||nvl(p.cpu_count,'')||','||nvl(p.processes,'')||','||nvl(p.sessions,'')||','||nvl(p.db_block_size,'')||','||
nvl(to_char(to_number(p.sga_target)/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||','||nvl(to_char(to_number(p.sga_max_size)/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||','||nvl(to_char(sa.bytes/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||','||nvl(to_char(to_number(p.pga_target)/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||',,'||nvl(to_char(pa.bytes/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||','||nvl(to_char(pa.maxbytes/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||','||nvl(to_char(to_number(p.memory_target)/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||','||nvl(to_char(to_number(p.memory_max_target)/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||',1,0,0,0,0,'||
df.cnt||','||df.autocnt||','||to_char(df.bytes/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,''')||','||to_char(df.bytes/power(1024,4),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,''')||','||to_char(df.maxbytes/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,''')||','||to_char(df.maxbytes/power(1024,4),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,''')||','||tf.cnt||','||tf.autocnt||','||nvl(to_char(tf.bytes/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||','||nvl(to_char(tf.maxbytes/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||','||rl.threads||','||rl.groups||','||rm.members||','||to_char(rl.bytes/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,''')||','||to_char(rl.minbytes/power(1024,2),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,''')||','||to_char(rl.maxbytes/power(1024,2),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,''')||','||cf.cnt||',"'||ai.uses_asm||'","'||replace(nvl(fr.name,p.fra_dest),'"','""')||'",'||nvl(to_char(nvl(fr.space_limit,to_number(p.fra_size))/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||','||nvl(to_char(fr.space_used/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||','||nvl(to_char(fr.space_reclaimable/power(1024,3),'FM9999999990D00','NLS_NUMERIC_CHARACTERS=''.,'''),'')||','||nvl(to_char(fr.number_of_files),'')||',"'||cs.dbcs||'","'||cs.ncs||'","'||tz.filename||'",'||tz.version||',"","","'||nvl(p.filesystemio_options,'')||'","'||nvl(p.disk_asynch_io,'')||'","'||nvl(p.db_writer_processes,'')||'","'||to_char(systimestamp,'YYYY-MM-DD HH24:MI:SS TZH:TZM')||'"'
from v$database d cross join gv$instance i join p on p.inst_id=i.inst_id
left join sa on sa.inst_id=i.inst_id left join pa on pa.inst_id=i.inst_id
cross join df cross join tf cross join rl cross join rm cross join cf cross join ai cross join cs cross join tz
left join fr on 1=1
order by i.instance_number;

spool off
set termout on
prompt
prompt Inventory CSV created:
prompt &output_file
prompt
exit success
